# =============================================================================
# Unified "baseline" requirements — pip/venv install for the 4-model committee
# =============================================================================
#
# Mirror of scripts/inference/setup_baseline_env.sh but for users who prefer
# Python venv over conda. Hosts dependencies for DT-Kinase, DrugBAN, GraphBAN
# and ConPLex inference in a single environment.
#
# CRITICAL — DO NOT INSTALL THIS FILE WITH `pip install -r` ALONE.
# PyTorch and DGL must be installed FIRST from their dedicated index URLs;
# otherwise pip pulls CPU-only PyTorch from PyPI (breaks torchvision::nms)
# or fails to find DGL wheels (HTTP 403).
#
# Recommended install (Linux + CUDA 12.1):
#
#     python -m venv env_baseline
#     source env_baseline/bin/activate
#     pip install --upgrade pip wheel setuptools
#
#     # 1. PyTorch (must come first, dedicated CUDA index URL)
#     pip install torch==2.4.1 torchvision torchaudio \
#         --index-url https://download.pytorch.org/whl/cu121
#
#     # 2. DGL (must come second, dedicated dgl-team URL; fall back to CPU
#     #    wheel from PyPI on macOS / no-CUDA hosts)
#     pip install dgl -f https://data.dgl.ai/wheels/torch-2.4/cu121/repo.html  # CUDA
#     #   OR
#     pip install dgl                                                          # CPU
#
#     # 3. Everything else (this file)
#     pip install -r requirements-baseline.txt
#
#     # 4. Patch DGL graphbolt __init__ (ABI workaround if CUDA wheel mismatch)
#     python -c "import dgl, os, pathlib; \
#         p = pathlib.Path(os.path.dirname(dgl.__file__))/'graphbolt/__init__.py'; \
#         p.exists() and p.write_text('# disabled — ABI mismatch')"
#
# CPU-only fallback (macOS, debug, CI):
#     pip install torch torchvision torchaudio          # standard PyPI build
#     pip install dgl                                    # standard CPU wheel
#     pip install -r requirements-baseline.txt
#
# After install, verify:
#     python -c "import torch, dgl, transformers, rdkit, esm; print('OK')"
#
# =============================================================================
